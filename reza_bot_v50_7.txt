def utc_now():
    return datetime.now(timezone.utc)


def parse_xt_timestamp(value):
    if value is None:
        return None

    try:
        numeric_value = float(value)
    except (TypeError, ValueError):
        return None

    if not math.isfinite(numeric_value):
        return None

    if numeric_value > 100000000000:
        numeric_value /= 1000.0

    try:
        return datetime.fromtimestamp(numeric_value, timezone.utc)
    except (OverflowError, OSError, ValueError):
        return None


def parse_utc_datetime(value):
    if value is None:
        return None

    if isinstance(value, datetime):
        if value.tzinfo is None:
            return value.replace(tzinfo=timezone.utc)
        return value.astimezone(timezone.utc)

    if isinstance(value, (int, float)):
        return parse_xt_timestamp(value)

    if isinstance(value, str):
        text = value.strip()
        if not text:
            return None

        try:
            numeric_value = float(text)
            return parse_xt_timestamp(numeric_value)
        except (TypeError, ValueError):
            pass

        if text.endswith(" UTC"):
            text = text[:-4]

        try:
            parsed = datetime.strptime(text, "%Y-%m-%d %H:%M:%S")
            return parsed.replace(tzinfo=timezone.utc)
        except ValueError:
            pass

        try:
            parsed = datetime.fromisoformat(text)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=timezone.utc)
            return parsed.astimezone(timezone.utc)
        except ValueError:
            pass

    return None


def format_utc(dt):
    if dt is None:
        return ""
    parsed = parse_utc_datetime(dt)
    if parsed is None:
        return ""
    return parsed.strftime("%Y-%m-%d %H:%M:%S UTC")


def restore_signal_datetime_fields(signal):
    if not isinstance(signal, dict):
        return signal

    for field in ("time", "source_candle"):
        if field not in signal:
            continue
        value = signal.get(field)
        parsed = parse_utc_datetime(value)
        if parsed is not None:
            signal[field] = parsed

    return signal

def session_start_for(dt):
    """محاسبه شروع سشن HH:30"""
    dt = parse_utc_datetime(dt)
    if dt is None:
        return None

    base = dt.replace(minute=0, second=0, microsecond=0)
    
    if dt.minute >= SESSION_MINUTE:
        return base.replace(minute=SESSION_MINUTE)
    else:
        return (base - timedelta(hours=1)).replace(minute=SESSION_MINUTE)


def next_session_start_after(dt):
    current = session_start_for(dt)
    if current is None:
        return None
    if current > dt:
        return current
    return current + timedelta(hours=1)


def session_end_for(start):
    start = parse_utc_datetime(start)
    if start is None:
        return None
    return start + timedelta(hours=1)


def candle_session_start(candle_time):
    return session_start_for(candle_time)

def atomic_write_json(path, data):
    ensure_data_dir()
    temporary_file = path + ".tmp"

    try:
        with open(temporary_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, default=str)
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temporary_file, path)

    except Exception as exc:
        log.error("Could not write JSON state %s: %s", path, exc)
        try:
            if os.path.exists(temporary_file):
                os.remove(temporary_file)
        except Exception:
            pass


def read_json_file(path, default):
    if not os.path.exists(path):
        return default

    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data

    except Exception as exc:
        log.warning("Could not read JSON state %s: %s", path, exc)
        return default

def restore_session_signal_lock():
    global session_state

    data = read_json_file(SESSION_STATE_FILE, {})

    if not isinstance(data, dict):
        return

    saved_start = data.get("start")
    if not saved_start:
        return

    saved_dt = parse_utc_datetime(saved_start)
    if saved_dt is None:
        return

    now = utc_now()
    current_session_start = session_start_for(now)

    if current_session_start != saved_dt:
        log.info("🔄 NEW SESSION DETECTED - RESETTING HYPER AND SIGNAL LOCK")
        log.info(f"   OLD SESSION: {format_utc(saved_dt)}")
        log.info(f"   NEW SESSION: {format_utc(current_session_start)}")

        if session_state is not None:
            session_state["signal_generated"] = False
            session_state["signal_id"] = None
            session_state["hyper_locked"] = False
            session_state["ready"] = False
            session_state["hyper"] = {
                "high": None,
                "low": None,
                "r": None,
                "source": "",
                "source_candle_time": None
            }
            session_state["start"] = current_session_start
            session_state["end"] = session_end_for(current_session_start)
            session_state["boundary_candle_time"] = None
            session_state["analysis_candles"] = 0
            
            if os.path.exists(SESSION_STATE_FILE):
                os.remove(SESSION_STATE_FILE)
            
            save_current_session_state()
        return

    if not data.get("signal_generated", False):
        return

    if session_state is None:
        return

    session_state["signal_generated"] = True
    session_state["signal_id"] = data.get("signal_id")
    session_state["hyper_locked"] = bool(data.get("hyper_locked", False))

    hyper = session_state.setdefault("hyper", {})
    if data.get("hyper_high") is not None:
        hyper["high"] = data.get("hyper_high")
    if data.get("hyper_low") is not None:
        hyper["low"] = data.get("hyper_low")
    if data.get("r") is not None:
        hyper["r"] = float(data.get("r"))

    log.info("SESSION SIGNAL LOCK RESTORED: %s", data.get("signal_id"))

def read_platform_queue():
    ensure_platform_queue_file()

    signals = []

    try:
        with open(PLATFORM_QUEUE_FILE, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue

                try:
                    signal = json.loads(line)
                    if not isinstance(signal, dict):
                        continue

                    signal = restore_signal_datetime_fields(signal)
                    signal_id = signal.get("signal_id")

                    if signal_id:
                        signals.append(signal)

                except Exception:
                    log.warning("Invalid platform queue line ignored.")

    except Exception as exc:
        log.error("Could not read platform queue: %s", exc)

    unique = []
    seen = set()

    for signal in signals:
        signal_id = str(signal.get("signal_id"))
        if signal_id in seen:
            continue
        seen.add(signal_id)
        unique.append(signal)

    return unique


def write_platform_queue(signals):
    ensure_platform_queue_file()

    temporary_file = PLATFORM_QUEUE_FILE + ".tmp"

    try:
        with open(temporary_file, "w", encoding="utf-8") as f:
            for signal in signals:
                f.write(json.dumps(signal, ensure_ascii=False, default=str) + "\n")
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temporary_file, PLATFORM_QUEUE_FILE)

    except Exception as exc:
        log.error("Could not write platform queue: %s", exc)
        try:
            if os.path.exists(temporary_file):
                os.remove(temporary_file)
        except Exception:
            pass


def refresh_pending_count():
    signals = read_platform_queue()

    with platform_lock:
        platform_state["pending_count"] = len(signals)

    return len(signals)


def queue_platform_signal(signal):
    signals = read_platform_queue()

    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))

    existing_ids = {str(x.get("signal_id")) for x in signals}

    if signal_id not in existing_ids:
        signals.append(signal)

    write_platform_queue(signals)

    with platform_lock:
        platform_state["pending_count"] = len(signals)


def remove_platform_signal(signal_id):
    signals = read_platform_queue()

    remaining = [
        x for x in signals
        if str(x.get("signal_id")) != str(signal_id)
    ]

    write_platform_queue(remaining)

    with platform_lock:
        platform_state["pending_count"] = len(remaining)

def platform_health_check():
    if not PLATFORM_ENABLED:
        with platform_lock:
            platform_state["online"] = False
            platform_state["last_error"] = "DISABLED"

        refresh_pending_count()
        return False

    try:
        response = platform_http.get(
            PLATFORM_BASE_URL + PLATFORM_HEALTH_PATH,
            timeout=PLATFORM_TIMEOUT
        )

        if response.status_code != 200:
            raise RuntimeError(f"HTTP {response.status_code}: {response.text[:500]}")

        try:
            data = response.json()
        except Exception:
            data = {}

        status = str(data.get("status", "ok")).lower()

        if status not in ("ok", "ready", "online", "healthy"):
            raise RuntimeError(f"Platform unhealthy: {status}")

        with platform_lock:
            platform_state["online"] = True
            platform_state["last_check"] = utc_now()
            platform_state["last_success"] = utc_now()
            platform_state["last_error"] = None

        refresh_pending_count()
        return True

    except Exception as exc:
        with platform_lock:
            platform_state["online"] = False
            platform_state["last_check"] = utc_now()
            platform_state["last_error"] = str(exc)

        refresh_pending_count()
        return False

def validate_platform_ack(data, sent_signal_id):
    if not isinstance(data, dict):
        return False

    returned_signal_id = data.get("signal_id")
    if returned_signal_id is None:
        return False

    if str(returned_signal_id) != str(sent_signal_id):
        log.error("INVALID PLATFORM ACK: signal_id mismatch. sent=%s returned=%s",
                  sent_signal_id, returned_signal_id)
        return False

    ack = bool(data.get("ack", False))
    status = str(data.get("status", "")).lower()
    already_received = bool(data.get("already_received", False))
    already_accepted = bool(data.get("already_accepted", False))

    if ack or already_received or already_accepted:
        return True

    if status in ("already_received", "already_accepted"):
        return True

    return False

            "WAITING_FOR_PLATFORM",
            "WAITING_FOR_PLATFORM",
            "PLATFORM_DISABLED"
        )
        return {"success": False, "status": "PLATFORM_DISABLED"}

    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))
    payload = build_platform_payload(signal)

    with platform_delivery_lock:
        try:
            response = platform_http.post(
                PLATFORM_BASE_URL + PLATFORM_SIGNAL_PATH,
                json=payload,
                timeout=PLATFORM_TIMEOUT
            )

            if response.status_code not in (200, 201, 202):
                raise RuntimeError(f"HTTP {response.status_code}: {response.text[:1000]}")

            try:
                data = response.json()
            except Exception:
                data = {}

            if not validate_platform_ack(data, signal_id):
                raise RuntimeError(f"Platform response did not contain valid ACK: {data}")

            with platform_lock:
                platform_state["online"] = True
                platform_state["last_success"] = utc_now()
                platform_state["last_error"] = None
                platform_state["last_ack_signal"] = signal_id

            remove_platform_signal(signal_id)

            update_signal_status(
                signal,
                "READY_FOR_EXECUTION",
                "ACKNOWLEDGED",
                "PLATFORM_ACK",
                response=data
            )

            return {"success": True, "status": data.get("status", "ACKNOWLEDGED"), "response": data}

        except Exception as exc:
            with platform_lock:
                platform_state["online"] = False
                platform_state["last_error"] = str(exc)

            queue_platform_signal(signal)

            update_signal_status(
                signal,
                "WAITING_FOR_PLATFORM",
                "WAITING_FOR_PLATFORM",
                "DELIVERY_FAILED_PENDING",
                response={"error": str(exc)}
            )

            return {"success": False, "status": "WAITING_FOR_PLATFORM", "error": str(exc)}

def get_ticker():
    url = f"{XT_BASE_URL}/future/market/v1/public/q/agg-ticker"
    params = {"symbol": SYMBOL}

    try:
        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()

    except Exception as exc:
        raise RuntimeError(f"XT ticker request failed for {SYMBOL}: {exc}")

    try:
        data = response.json()

    except Exception as exc:
        raise RuntimeError(f"XT returned non-JSON ticker for {SYMBOL}: {response.text[:1000]}") from exc

    if not isinstance(data, dict):
        raise RuntimeError(f"XT returned invalid ticker for {SYMBOL}: {repr(data)[:1500]}")

    result = data.get("result")

    if isinstance(result, list):
        if not result:
            raise RuntimeError(f"XT returned empty ticker for {SYMBOL}: {repr(data)[:1500]}")
        result = result[0]

    elif not isinstance(result, dict):
        raise RuntimeError(f"XT returned invalid ticker result for {SYMBOL}: {repr(data)[:1500]}")

    last = result.get("c") if result.get("c") is not None else result.get("last")
    if last is None:
        last = result.get("lastPrice") if result.get("lastPrice") is not None else result.get("price")

    if last is None:
        raise RuntimeError(f"XT ticker has no last price for {SYMBOL}: {repr(data)[:1500]}")

    try:
        last = float(last)

    except (TypeError, ValueError) as exc:
        raise RuntimeError(f"XT ticker returned invalid last price for {SYMBOL}: {repr(last)}") from exc

    if not math.isfinite(last) or last <= 0:
        raise RuntimeError(f"XT ticker returned invalid last price for {SYMBOL}: {repr(last)}")

    ts = result.get("ts") if result.get("ts") is not None else result.get("timestamp")
    if ts is None:
        ts = result.get("time") if result.get("time") is not None else result.get("t")

    return {"last": last, "ts": ts, "raw": data}

# ATR CALCULATION
# ============================================================================

def calculate_atr(candles, period=5):
    if len(candles) < 2:
        return None
    
    available_period = min(period, len(candles) - 1)
    
    if available_period < 1:
        return None
    
    tr_list = []
    for i in range(1, len(candles)):
        high = candles[i]['high']
        low = candles[i]['low']
        prev_close = candles[i-1]['close']
        
        tr = max(high - low, abs(high - prev_close), abs(low - prev_close))
        tr_list.append(tr)
    
    if not tr_list:
        return None
    
    atr = sum(tr_list[-available_period:]) / available_period
    
    log.info(f"📊 ATR CALCULATED: period={available_period}/{period}, atr={atr:.4f}")
    
    return atr

def setup_session_locked_levels(state, first_closed_candle):
    if state is None or first_closed_candle is None:
        return False

    if first_closed_candle.get("confirmed", 0) != 1:
        log.error("SESSION INIT FAILED: first candle is not closed.")
        return False

    candle_time = first_closed_candle.get("time")
    if candle_time is None:
        log.error("SESSION INIT FAILED: first candle time missing.")
        return False

    expected_session_start = session_start_for(candle_time)
    if candle_time != expected_session_start:
        log.error("SESSION INIT FAILED: candle is not HH:30 session candle | candle=%s | session_start=%s",
                  format_utc(candle_time), format_utc(expected_session_start))
        return False

    if state.get("hyper_locked", False):
        log.warning("⚠️ HYPER ALREADY LOCKED, skipping setup")
        return True

    hyper_high = float(first_closed_candle["high"])
    hyper_low = float(first_closed_candle["low"])
    r = hyper_high - hyper_low

    if hyper_high < hyper_low:
        log.error("SESSION INIT FAILED: invalid Hyper range | high=%s low=%s", hyper_high, hyper_low)
        return False

    # ================================================================
    # ✅ فیلتر حجم حذف شد
    # ================================================================

    # ================================================================
    # بررسی R
    # ================================================================
    if r < MIN_R_VALUE:
        log.error(f"❌ R FILTER FAILED: r={r:.4f} < {MIN_R_VALUE}")
        return False

    # ================================================================
    # بررسی بدنه کندل
    # ================================================================
    body = abs(first_closed_candle["close"] - first_closed_candle["open"])
    total_range = hyper_high - hyper_low
    if total_range > 0:
        body_percent = (body / total_range) * 100
    else:
        body_percent = 0

    if body_percent < MIN_BODY_PERCENT:
        log.error(f"❌ BODY FILTER FAILED: body_percent={body_percent:.2f}% < {MIN_BODY_PERCENT}%")
        return False

    log.info(f"✅ ALL FILTERS PASSED: r={r:.4f}, body={body_percent:.2f}%")

    # ================================================================
    # قفل کردن هایپر
    # ================================================================
    state["ready"] = True
    state["analysis_candles"] = 1
    state["structure"] = {}
    state["hyper"] = {
        "high": hyper_high,
        "low": hyper_low,
        "r": r,
        "source": "SESSION_FIRST_CANDLE_LOCKED",
        "source_candle_time": candle_time
    }
    state["hyper_locked"] = True
    state["boundary_candle_time"] = candle_time

    save_session(state)
    save_current_session_state()

    log.info("✅ SESSION READY | start=%s | first_candle=%s | hyper_high=%s | hyper_low=%s | R=%s | body=%s%%",
             format_utc(state["start"]), format_utc(candle_time), hyper_high, hyper_low, r, body_percent)

    return True

def rollover_session_if_needed(new_start, first_closed_candle=None):
    global session_state, last_session_start, startup_waiting_for_session

    if last_session_start != new_start:
        log.info("🔄 SESSION ROLLOVER: %s -> %s", last_session_start, new_start)

        new_state = prepare_session(
            new_start,
            boundary_candle_time=first_closed_candle["time"] if first_closed_candle else None
        )

        new_state["hyper"] = {
            "high": None,
            "low": None,
            "r": None,
            "source": "",
            "source_candle_time": None
        }
        new_state["hyper_locked"] = False
        new_state["ready"] = False
        new_state["signal_generated"] = False
        new_state["signal_id"] = None

        session_state = new_state
        last_session_start = new_start
        startup_waiting_for_session = False

        if os.path.exists(SESSION_STATE_FILE):
            os.remove(SESSION_STATE_FILE)

        save_current_session_state()

        if first_closed_candle:
            return setup_session_locked_levels(session_state, first_closed_candle)

        return True

    if first_closed_candle and first_closed_candle.get("confirmed", 0) == 1:
        candle_session = session_start_for(first_closed_candle["time"])
        if candle_session == new_start:
            log.info("🔄 RE-INITIALIZING SESSION HYPER FOR: %s", format_utc(new_start))
            return setup_session_locked_levels(session_state, first_closed_candle)

    return True

# SIGNAL DETECTION
# ============================================================================

def detect_signal(live_price, live_candle, session_state):
    if live_price is None or live_candle is None or session_state is None:
        return None

    if not session_state.get("ready", False) or not session_state.get("hyper_locked", True):
        return None

    if session_state.get("signal_generated", False):
        return None

    hyper = session_state.get("hyper")
    if not isinstance(hyper, dict):
        return None

    hyper_high = hyper.get("high")
    hyper_low = hyper.get("low")
    r = hyper.get("r")

    if hyper_high is None or hyper_low is None or r is None:
        return None

    price = float(live_price)

    if price >= float(hyper_high):
        return {
            "direction": "LONG",
            "entry": float(hyper_high),
            "touch_price": price,
            "candle": live_candle,
            "hyper_high": float(hyper_high),
            "hyper_low": float(hyper_low),
            "r": float(r),
            "touch_type": "HIGH_TOUCH"
        }

    if price <= float(hyper_low):
        return {
            "direction": "SHORT",
            "entry": float(hyper_low),
            "touch_price": price,
            "candle": live_candle,
            "hyper_high": float(hyper_high),
            "hyper_low": float(hyper_low),
            "r": float(r),
            "touch_type": "LOW_TOUCH"
        }

    return None

# BUILD REFERENCE SIGNAL
# ============================================================================

def build_reference_signal(strategy_signal, breakout_candle):
    if strategy_signal is None or breakout_candle is None:
        return None

    direction = strategy_signal.get("direction")
    if direction not in ("LONG", "SHORT"):
        return None

    if direction == "LONG":
        entry = float(strategy_signal["hyper_high"])
    else:
        entry = float(strategy_signal["hyper_low"])

    breakout_time = breakout_candle["time"]
    signal_id = f"{BOT_ID}-{direction}-{breakout_time.strftime('%Y%m%d%H%M')}"

    hyper_high = float(strategy_signal["hyper_high"])
    hyper_low = float(strategy_signal["hyper_low"])
    r = float(hyper_high - hyper_low)

    atr = None
    stop_distance = r
    
    if USE_ATR_STOP:
        atr = calculate_atr(primary_candles, ATR_PERIOD)
        
        if atr is not None:
            stop_distance = atr * ATR_MULTIPLIER
            log.info(f"📊 ATR CALCULATION: atr={atr:.4f}, multiplier={ATR_MULTIPLIER}, stop_distance={stop_distance:.4f}")
        else:
            log.warning("⚠️ ATR CALCULATION FAILED (not enough candles), using R for stop")
    
    min_stop = r * 0.5
    max_stop = r * 2.0
    
    if stop_distance < min_stop:
        stop_distance = min_stop
        log.warning(f"⚠️ STOP DISTANCE TOO SMALL, adjusted to {min_stop:.4f}")
    elif stop_distance > max_stop:
        stop_distance = max_stop
        log.warning(f"⚠️ STOP DISTANCE TOO LARGE, adjusted to {max_stop:.4f}")
    
    if direction == "LONG":
        initial_stop = entry - stop_distance
        initial_tp = entry + (r * 1.5)
    else:
        initial_stop = entry + stop_distance
        initial_tp = entry - (r * 1.5)

    signal = {
        "signal_id": signal_id,
        "time": utc_now(),
        "source_candle": breakout_time,
        "symbol": SYMBOL,
        "reference_market": "XT XRPUSDT FUTURES",
        "direction": direction,
        "reference_entry": float(entry),
        "hyper_high": float(hyper_high),
        "hyper_low": float(hyper_low),
        "r": float(r),
        "initial_risk": float(r),
        "initial_stop": float(initial_stop),
        "initial_tp": float(initial_tp),
        "tp_r_multiple": 1.5,
        "entry_type": "HYPER_TOUCH",
        "touch_price": float(strategy_signal["touch_price"]),
        "touch_type": strategy_signal.get("touch_type", ""),
        "bot_id": BOT_ID,
        "atr_used": float(atr) if atr is not None else None,
        "stop_distance": float(stop_distance),
        "candles_used": len(primary_candles)
    }

    log.info("SIGNAL CREATED | id=%s | direction=%s | entry=%s | stop=%s | tp=%s | r=%s | atr=%s | candles=%s",
             signal_id, direction, signal["reference_entry"], 
             signal["initial_stop"], signal["initial_tp"], 
             signal["r"], signal["atr_used"], len(primary_candles))

    return signal

# LIVE PRICE PROCESSOR
# ============================================================================

def process_live_market(market):
    global current_candle, last_bucket, last_price, market_state, primary_candles
    global session_state, last_session_start, startup_waiting_for_session

    price = float(market["last"])
    now = parse_utc_datetime(market.get("ts"))

    if now is None:
        now = utc_now()

    last_price = price
    market_state = market
    market_state["ts"] = now

    save_price(now, market["last"], market.get("mark"), market.get("index"), market.get("funding"))

    bucket = bucket_5m(now)

    if current_candle is None:
        current_candle = start_new_candle(bucket, price)
        last_bucket = bucket
        return

    if bucket == last_bucket:
        update_candle(current_candle, price)

    else:
        previous = close_candle(current_candle)
        previous["confirmed"] = 1

        if not primary_candles or previous["time"] > primary_candles[-1]["time"]:
            primary_candles.append(previous)

            if len(primary_candles) > MAX_PRIMARY_CANDLES:
                primary_candles = primary_candles[-MAX_PRIMARY_CANDLES:]

                  f"O={previous['open']:.4f} H={previous['high']:.4f} L={previous['low']:.4f} C={previous['close']:.4f} | "
                  f"samples={previous.get('samples', 0)} | closed_history={len(primary_candles)}")

        closed_session_start = session_start_for(previous["time"])
        is_first_session_candle = (previous["time"] == closed_session_start)

        if is_first_session_candle:
            print()
            print("=" * 78)
            print(f"HH:30 CANDLE CLOSED - {format_utc(closed_session_start)}")
            print(f"HH:30 HIGH   : {previous['high']:.4f}")
            print(f"HH:30 LOW    : {previous['low']:.4f}")
            print(f"R            : {previous['high'] - previous['low']:.4f}")
            print("ACTION       : LOCK HYPER")
            print("=" * 78)

            if last_session_start == closed_session_start:
                log.warning(f"🔄 RESETTING last_session_start: {last_session_start} -> None")
                last_session_start = None

            rollover_session_if_needed(closed_session_start, previous)

        current_candle = start_new_candle(bucket, price)
        last_bucket = bucket

    # ================================================================
    # بررسی سیگنال
    # ================================================================
    if (session_state is not None and
        session_state.get("ready", False) and
        session_state.get("hyper_locked", False) and
        not session_state.get("signal_generated", False)):

        strategy_signal = detect_signal(price, current_candle, session_state)

        if strategy_signal:
            signal = build_reference_signal(strategy_signal, current_candle)

            if signal:
                emit_signal(signal)

class WebhookHandler(BaseHTTPRequestHandler):

    def log_message(self, format, *args):
        return

    def send_json(self, status_code, payload):
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self.send_response(status_code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        if self.path == WEBHOOK_PATH:
            try:
                length = int(self.headers.get("Content-Length", "0"))
                raw = self.rfile.read(length)
                data = json.loads(raw.decode("utf-8"))

                event = data.get("event")
                signal_id = data.get("signal_id")

                log.info("📡 WEBHOOK RECEIVED: event=%s signal_id=%s", event, signal_id)

                append_webhook_log(data)

                print()
                print("=" * 78)
                print("📡 WEBHOOK RECEIVED FROM EXECUTION PLATFORM")
                print("=" * 78)
                print(f"EVENT           : {event}")
                print(f"SIGNAL ID       : {signal_id}")
                print(f"DIRECTION       : {data.get('direction')}")
                print(f"ENTRY PRICE     : {data.get('entry_price')}")
                print(f"EXIT PRICE      : {data.get('exit_price')}")
                print(f"P&L             : {data.get('pnl')} USDT")
                print(f"P&L %           : {data.get('pnl_percent')}%")
                print(f"REASON          : {data.get('reason')}")
                print(f"FINAL STEP      : {data.get('final_step')} ({(data.get('total_r', 0)):.1f}R)")
                print(f"CLOSE TIME      : {data.get('close_time')}")
                print("=" * 78)
                print()

                self.send_json(200, {
                    "status": "OK",
                    "message": "Webhook received",
                    "signal_id": signal_id
                })

            except Exception as exc:
                log.error("WEBHOOK HANDLER ERROR: %s", exc)
                self.send_json(400, {"status": "ERROR", "error": str(exc)})

        else:
            self.send_json(404, {"error": "NOT_FOUND"})

    def do_GET(self):
        if self.path == "/health":
            self.send_json(200, {
                "bot": f"REZA BOT {BOT_ID}",
                "version": VERSION,
                "project": PROJECT,
                "symbol": DISPLAY_SYMBOL,
                "status": "ONLINE"
            })
        else:
            self.send_json(404, {"error": "NOT_FOUND"})


def append_webhook_log(data):
    with open(WEBHOOK_LOG_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(data, ensure_ascii=False, default=str) + "\n")


def start_webhook_server():
    try:
        server = ThreadingTCPServer((WEBHOOK_HOST, WEBHOOK_PORT), WebhookHandler)
        server.allow_reuse_address = True
        server.daemon_threads = True

        thread = threading.Thread(target=server.serve_forever, daemon=True, name="WEBHOOK-SERVER")
        thread.start()

        print()
        print("=" * 78)
        print("🌐 WEBHOOK SERVER STARTED")
        print("=" * 78)
        print(f"WEBHOOK LISTENER : http://{WEBHOOK_HOST}:{WEBHOOK_PORT}{WEBHOOK_PATH}")
        print(f"HEALTH           : http://{WEBHOOK_HOST}:{WEBHOOK_PORT}/health")
        print("=" * 78)

        return True

    except OSError as exc:
        if getattr(exc, "errno", None) == 98:
            print()
            print("=" * 78)
            print("⚠️ WEBHOOK SERVER: PORT 8765 IS ALREADY IN USE")
            print("=" * 78)
            print("Existing webhook server may be running.")
            print("=" * 78)
            return False
        raise
